Google Scholar Paper Extractor

Instructions:

  1. Open the Google Scholar profile: https://scholar.google.com/citations?user=hLBNWsYAAAAJ&hl=en
  2. Open browser console (F12 or Cmd+Option+I)
  3. Copy and paste the script below into the console
  4. Press Enter and copy the output
  5. Paste the output in the textarea below and click "Format Papers"

Console Script:

// Copy this entire script to browser console on Google Scholar page (function() { const papers = []; const rows = document.querySelectorAll('tr.gsc_a_tr'); rows.forEach((row, idx) => { const titleLink = row.querySelector('a.gsc_a_at'); if (!titleLink) return; const title = titleLink.textContent.trim(); const link = titleLink.href || `https://scholar.google.com${titleLink.getAttribute('href')}`; const authorsVenue = row.querySelector('div.gs_gray'); let authors = '', venue = ''; if (authorsVenue) { const text = authorsVenue.textContent.trim(); const parts = text.split(' - '); authors = parts[0]?.trim() || ''; venue = parts[1]?.trim() || ''; } const yearSpan = row.querySelector('span.gsc_a_y'); const year = yearSpan ? parseInt(yearSpan.textContent.trim()) : null; papers.push({ id: idx + 1, title: title, authors: authors, venue: venue, year: year, link: link }); }); console.log(JSON.stringify(papers, null, 2)); return papers; })();
